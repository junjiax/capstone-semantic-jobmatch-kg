// Run app
// Redirect to work dir
cd fastapi

// Window
venv\Scripts\activate

// Linux 
source venv/Scripts/activate

// Chỉ định tên file cấu hình khác:
docker compose -f file-name.yml up -d

// Build lại toàn bộ image trước khi khởi động (nếu bạn đã sửa mã nguồn hoặc Dockerfile):
docker compose up --build -d

// Dừng và xóa toàn bộ các container đã tạo bởi file compose:
docker compose down

// Run fastapi local
uvicorn app.main:app --reload --host 127.0.0.1 --port 8080


// Xem bảng trong postgressql
$ docker compose exec web python -c "from sqlalchemy import inspect; from app.core.database import engine; print(inspect(engine).get_table_names())"
